apiVersion: networking.k8s.io/v1

metadata:
  name: isolate-juice-shop-ingress
  namespace: default
spec:
  podSelector:
    matchLabels:
      app: juice-shop
  policyTypes:
  - Ingress
  ingress:
  - from:
    - namespaceSelector:
        matchLabels:

    - protocol: TCP
      port: 3000
# terraform/security_groups.tf (Restricts 0.0.0.0/0 Public Access)
resource "aws_security_group_rule" "lockdown_port_3000" {
  type              = "ingress"
  from_port         = 3000
  to_port           = 3000
  protocol          = "tcp"
  cidr_blocks       = ["10.0.0.0/16"] # Restricted Internal VPC only
